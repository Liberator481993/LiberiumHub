-- Executors expose different environment APIs; do not abort UI startup when getgenv is absent.
local function resolveExecutorEnvironment()
    local ok, environment = pcall(function()
        if type(getgenv) == "function" then return getgenv() end
        return _G
    end)
    if ok and type(environment) == "table" then return environment end
    return {}
end
local GEN_ENV = resolveExecutorEnvironment()

-- Resolve optional executor functions without assuming Delta and Arceus X expose them
-- in the same environment table. Missing APIs stay nil; UI startup never depends on them.
local function resolveExecutorFunction(apiName)
    local candidate = nil
    pcall(function()
        candidate = GEN_ENV[apiName]
    end)
    if type(candidate) == "function" then return candidate end

    pcall(function()
        local currentEnvironment = getfenv(0)
        if type(currentEnvironment) == "table" then
            candidate = currentEnvironment[apiName]
        end
    end)
    if type(candidate) == "function" then return candidate end

    pcall(function()
        candidate = _G[apiName]
    end)
    if type(candidate) == "function" then return candidate end
    return nil
end

local function callExecutorName(apiName)
    local api = resolveExecutorFunction(apiName)
    if not api then return nil end
    local ok, value = pcall(api)
    if ok and type(value) == "string" and value ~= "" then
        return value
    end
    return nil
end

local KILLER_ROLES = { murderer = true, zombie = true, killer = true }
local function normalizeRoleName(role)
    if type(role) == "table" then
        role = role.Role or role.role or role.RoleName or role.roleName or role.PlayerRole or role.playerRole or role.Name or role.name or role.Value or role.value
    end
    if typeof(role) == "Instance" then role = role.Name end
    if type(role) ~= "string" then return nil end
    local key = string.lower(role):gsub("[%s_%-]", "")
    if key:find("murderer", 1, true) or key == "murder" then return "murderer" end
    if key:find("sheriff", 1, true) then return "sheriff" end
    if key:find("hero", 1, true) then return "hero" end
    if key:find("innocent", 1, true) then return "innocent" end
    if key:find("survivor", 1, true) then return "survivor" end
    if key:find("zombie", 1, true) then return "zombie" end
    if key:find("killer", 1, true) then return "killer" end
    return nil
end
local function getPlayerRoleData(tbl, player)
    if type(tbl) ~= "table" or not player then return nil end
    local uid = player.UserId
    local uidText = tostring(uid)
    local data = tbl[uid] or tbl[uidText] or tbl[player.Name] or tbl[player] 
    if data ~= nil then return data end
    local wantedName = string.lower(player.Name)
    for key, value in pairs(tbl) do
        if string.lower(tostring(key)) == wantedName or tostring(key) == uidText then return value end
        if type(value) == "table" then
            local valueId = value.UserId or value.userId or value.UserID or value.userID
            local valueName = value.Name or value.name or value.PlayerName or value.playerName
            if (valueId ~= nil and tostring(valueId) == uidText) or (type(valueName) == "string" and string.lower(valueName) == wantedName) then
                return value
            end
        end
    end
    return nil
end
local function readExplicitRole(container)
    if not container then return nil end
    local fieldRole = nil
    local attributeNames = {"Role", "role", "RoleName", "roleName", "PlayerRole", "playerRole", "MM2Role", "CurrentRole"}
    for _, attributeName in ipairs(attributeNames) do
        local ok, attributeValue = pcall(function() return container:GetAttribute(attributeName) end)
        if ok then
            local role = normalizeRoleName(attributeValue)
            if role and role ~= "innocent" then return role end
            if role then fieldRole = role end
        end
    end
    -- Specific role flags must beat a generic/default Innocent value. Hero is a real
    -- MM2 round role and must retain its own yellow ESP instead of inheriting green.
    local priorityGroups = {
        {"murderer", {"IsMurderer", "isMurderer", "Murderer", "murderer", "IsKiller", "isKiller", "Killer", "killer"}},
        {"sheriff", {"IsSheriff", "isSheriff", "Sheriff", "sheriff"}},
        {"hero", {"IsHero", "isHero", "Hero", "hero"}},
        {"zombie", {"IsZombie", "isZombie", "Zombie", "zombie"}},
        {"survivor", {"IsSurvivor", "isSurvivor", "Survivor", "survivor"}},
        {"innocent", {"IsInnocent", "isInnocent", "Innocent", "innocent"}}
    }
    local fallbackFlag = nil
    for _, group in ipairs(priorityGroups) do
        local roleName, flags = group[1], group[2]
        for _, flagName in ipairs(flags) do
            local ok, flagValue = pcall(function() return container:GetAttribute(flagName) end)
            if ok and flagValue == true then
                if roleName ~= "innocent" then return roleName end
                fallbackFlag = fallbackFlag or roleName
            end
        end
    end
    local childRole = nil
    for _, child in ipairs(container:GetChildren()) do
        if child:IsA("BoolValue") and child.Value == true then
            for _, group in ipairs(priorityGroups) do
                for _, flagName in ipairs(group[2]) do
                    if child.Name == flagName then
                        if group[1] ~= "innocent" then return group[1] end
                        fallbackFlag = fallbackFlag or group[1]
                    end
                end
            end
        end
        local childName = string.lower(child.Name)
        if childName == "role" or childName == "rolename" or childName == "playerrole" or childName == "mm2role" then
            local value
            if child:IsA("StringValue") then value = child.Value
            elseif child:IsA("ObjectValue") and child.Value then value = child.Value.Name end
            local role = normalizeRoleName(value)
            if role and role ~= "innocent" then return role end
            if role then childRole = role end
        end
    end
    return childRole or fieldRole or fallbackFlag
end
local ROLE_FLAG_GROUPS = {
    { role = "murderer", flags = {"IsMurderer", "isMurderer", "Murderer", "murderer", "IsKiller", "isKiller", "Killer", "killer"} },
    { role = "sheriff", flags = {"IsSheriff", "isSheriff", "Sheriff", "sheriff"} },
    { role = "hero", flags = {"IsHero", "isHero", "Hero", "hero"} },
    { role = "innocent", flags = {"IsInnocent", "isInnocent", "Innocent", "innocent"} },
    { role = "survivor", flags = {"IsSurvivor", "isSurvivor", "Survivor", "survivor"} },
    { role = "zombie", flags = {"IsZombie", "isZombie", "Zombie", "zombie"} },
}
local ROLE_FIELDS = {"Role", "role", "RoleName", "roleName", "PlayerRole", "playerRole", "CurrentRole", "currentRole", "TeamRole", "teamRole"}
local ROLE_NESTED_FIELDS = {"Data", "data", "PlayerData", "playerData", "State", "state"}
local function roleFromFlags(data)
    if type(data) ~= "table" then return nil end
    for _, group in ipairs(ROLE_FLAG_GROUPS) do
        for _, flag in ipairs(group.flags) do
            if data[flag] == true then return group.role end
        end
    end
    for _, nestedKey in ipairs(ROLE_NESTED_FIELDS) do
        local nested = data[nestedKey]
        if type(nested) == "table" then
            for _, group in ipairs(ROLE_FLAG_GROUPS) do
                for _, flag in ipairs(group.flags) do
                    if nested[flag] == true then return group.role end
                end
            end
        end
    end
    return nil
end
local function roleFromFields(data)
    if type(data) ~= "table" then return normalizeRoleName(data) end
    local fallback = nil
    for _, field in ipairs(ROLE_FIELDS) do
        local role = normalizeRoleName(data[field])
        if role and role ~= "innocent" then return role end
        if role then fallback = fallback or role end
    end
    for _, nestedKey in ipairs(ROLE_NESTED_FIELDS) do
        local nested = data[nestedKey]
        if type(nested) == "table" then
            for _, field in ipairs(ROLE_FIELDS) do
                local role = normalizeRoleName(nested[field])
                if role and role ~= "innocent" then return role end
                if role then fallback = fallback or role end
            end
        end
    end
    return fallback
end
local function resolvePlayerRole(tbl, player)
    if not player then return nil, nil end
    local data = getPlayerRoleData(tbl, player)
    local fieldRole = roleFromFields(data)
    local flaggedRole = roleFromFlags(data)
    local explicitRole = readExplicitRole(player)
    local characterRole = player.Character and readExplicitRole(player.Character) or nil
    -- Use a specific role over a generic Innocent marker, since Hero can inherit
    -- Innocent metadata from the round payload while retaining its own role flag.
    if fieldRole and fieldRole ~= "innocent" then return fieldRole, data end
    if flaggedRole and flaggedRole ~= "innocent" then return flaggedRole, data end
    if explicitRole and explicitRole ~= "innocent" then return explicitRole, data end
    if characterRole and characterRole ~= "innocent" then return characterRole, data end
    return fieldRole or flaggedRole or explicitRole or characterRole, data
end
local function isKillerRole(role) return KILLER_ROLES[normalizeRoleName(role)] == true end

local InterceptSolver = (function() local API = {} function API.solve(targetPos, targetVel, originPos, speed) if type(speed) ~= "number" or speed <= 0 then return nil, nil end local speedSq = speed * speed local velSq = targetVel:Dot(targetVel) local D = targetPos - originPos local a = velSq - speedSq local h = D:Dot(targetVel) local c = D:Dot(D) if c < 1e-6 then return targetPos, 0 end local t if math.abs(a) < 1e-6 then if math.abs(h) < 1e-6 or h >= 0 then return nil, nil end t = -c / (2 * h) else local disc = h * h - a * c if disc < 0 then return nil, nil end local root = math.sqrt(disc) local t1 = (-h + root) / a local t2 = (-h - root) / a local best = math.huge if t1 > 0 and t1 < best then best = t1 end if t2 > 0 and t2 < best then best = t2 end if best == math.huge then return nil, nil end t = best end if not t or t ~= t or t <= 0 or t == math.huge then return nil, nil end return targetPos + targetVel * t, t end return API end)()

local RoleHUD = (function()
    local API = {}
    API.__index = API
    local COLORS = {
        murderer = Color3.fromRGB(255, 50, 50), killer = Color3.fromRGB(255, 50, 50), zombie = Color3.fromRGB(255, 50, 50),
        sheriff = Color3.fromRGB(50, 120, 255), hero = Color3.fromRGB(255, 230, 0),
        innocent = Color3.fromRGB(65, 220, 105), survivor = Color3.fromRGB(65, 220, 105)
    }

    local function destroyLayer(store, player)
        local layer = store[player]
        if layer then pcall(function() layer:Destroy() end) end
        store[player] = nil
    end

    function API.new(provider)
        return setmetatable({
            _provider = provider,
            _hl = {},             -- outline-only Highlights
            _fillHL = {},         -- fill-only Highlights; isolated from Outline
            _state = {},
            _conns = {},
            _running = false,
            _container = nil,
            _runToken = 0
        }, API)
    end

    function API:_ensureLayer(store, player, name)
        local layer = store[player]
        if not layer or not layer.Parent then
            layer = Instance.new("Highlight")
            layer.Name = name
            layer.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            layer.Parent = self._container
            store[player] = layer
        end
        return layer
    end

    function API:_apply(player, char, color)
        if not self._container or not self._container.Parent then return end

        local outline = self._hl[player]
        if CFG.ESP_Outline then
            outline = self:_ensureLayer(self._hl, player, "LiberiumRoleOutline")
        end
        if outline then
            outline.Adornee = char
            outline.FillTransparency = 1
            outline.OutlineColor = color
            outline.OutlineTransparency = CFG.ESP_Outline and 0 or 1
            outline.Enabled = CFG.ESP_Outline
        end

        local fill = self._fillHL[player]
        if CFG.ESP_Fill then
            fill = self:_ensureLayer(self._fillHL, player, "LiberiumRoleFill")
        end
        if fill then
            fill.Adornee = char
            fill.FillColor = color
            fill.FillTransparency = CFG.ESP_Fill and 0.42 or 1
            fill.OutlineTransparency = 1
            fill.Enabled = CFG.ESP_Fill
        end

        self._state[player] = color
    end

    function API:_update(player, tbl)
        local role, data = resolvePlayerRole(tbl, player)
        local char = player.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local dead = (type(data) == "table" and (data.Dead == true or data.IsDead == true or data.isDead == true))
            or not hum or hum.Health <= 0
        local color = role and COLORS[role]
        if not color or dead or not char or not char.Parent then
            local outline, fill = self._hl[player], self._fillHL[player]
            if outline then outline.Enabled = false; outline.Adornee = nil end
            if fill then fill.Enabled = false; fill.Adornee = nil end
            self._state[player] = nil
            return
        end
        self:_apply(player, char, color)
    end

    function API:_step()
        local ok, tbl = pcall(self._provider)
        if not ok or type(tbl) ~= "table" then return end
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then self:_update(player, tbl) end
        end
    end

    function API:Start()
        if self._running then return end
        self._running = true
        self._runToken += 1
        local token = self._runToken
        if not self._container or not self._container.Parent then
            local stale = Workspace:FindFirstChild("LibHUD")
            if stale then pcall(function() stale:Destroy() end) end
            self._container = Instance.new("Folder")
            self._container.Name = "LibHUD"
            self._container.Parent = Workspace
        end
        table.insert(self._conns, Players.PlayerRemoving:Connect(function(player)
            destroyLayer(self._hl, player)
            destroyLayer(self._fillHL, player)
            self._state[player] = nil
        end))
        task.spawn(function()
            while self._running and self._runToken == token and isRunning do
                self:_step()
                task.wait(0.2)
            end
        end)
    end

    function API:Stop()
        self._running = false
        self._runToken += 1
        for _, connection in ipairs(self._conns) do pcall(function() connection:Disconnect() end) end
        table.clear(self._conns)
        for player in pairs(self._hl) do destroyLayer(self._hl, player) end
        for player in pairs(self._fillHL) do destroyLayer(self._fillHL, player) end
        table.clear(self._state)
    end

    function API:Destroy()
        self:Stop()
        if self._container then pcall(function() self._container:Destroy() end); self._container = nil end
    end

    function API:Refresh()
        local ok, roleData = pcall(self._provider)
        if ok and type(roleData) == "table" then
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer then self:_update(player, roleData) end
            end
            return
        end
        for player, color in pairs(self._state) do
            local char = player.Character
            if char and char.Parent and color then
                self:_apply(player, char, color)
            else
                local outline, fill = self._hl[player], self._fillHL[player]
                if outline then outline.Enabled = false end
                if fill then fill.Enabled = false end
            end
        end
    end

    return API
end)()

local function applyToEffect(effect, values)
    local previousTween = activeEffectTweens[effect]
    if previousTween then
        activeEffectTweens[effect] = nil
        pcall(function() previousTween:Cancel() end)
    end

    local tweenGoals = {}
    for property, value in pairs(values) do
        if property == "Enabled" then
            if value then effect.Enabled = true end
        else
            tweenGoals[property] = value
        end
    end

    local activeTween = TweenService:Create(effect, TRANSITION, tweenGoals)
    activeEffectTweens[effect] = activeTween
    activeTween.Completed:Connect(function(playbackState)
        if activeEffectTweens[effect] ~= activeTween then return end
        activeEffectTweens[effect] = nil
        if values.Enabled == false and playbackState == Enum.PlaybackState.Completed then
            pcall(function() effect.Enabled = false end)
        end
    end)
    activeTween:Play()
end

    if not preset then return false end
    for key, values in pairs(preset) do
        pcall(function() applyToEffect(getEffect(key), values) end)
    end
    return true
end

function AM.CleanupShaders()

        local activeTween = activeEffectTweens[effect]

        end

        if effect and effect.Parent then
            if createdEffects[effect] then
                pcall(function() effect:Destroy() end)
            else
                local snapshot = originalEffectValues[effect]
                if snapshot then
                    for property, value in pairs(snapshot) do
                        pcall(function() effect[property] = value end)
                    end
                end
            end
        end

        effectByKey[key] = nil
    end
    table.clear(originalEffectValues)
    table.clear(createdEffects)
    table.clear(activeEffectTweens)

                state.character = nil
            end
        end)
    end
    state.type = auraType
    if player.Character then
        state.character = player.Character
        task.spawn(buildAura, player.Character, auraType, player, state)
    end
    return true
end
function AM.RemoveAura(player)
    local state = auraState[player]
    if state then
        if state.charConn then pcall(function() state.charConn:Disconnect() end) end
        if state.removeConn then pcall(function() state.removeConn:Disconnect() end) end
        destroyAuraInstances(state.character)
        auraState[player] = nil
    end
    destroyAuraInstances(player and player.Character)
end
function AM.ClearAllAuras() local players = {} for player in pairs(auraState) do table.insert(players, player) end for _, player in ipairs(players) do AM.RemoveAura(player) end end playerRemovingConnection = Players.PlayerRemoving:Connect(function(player)
    local state = auraState[player]
    if state then
        if state.charConn then pcall(function() state.charConn:Disconnect() end) end
        if state.removeConn then pcall(function() state.removeConn:Disconnect() end) end
        destroyAuraInstances(state.character)
        auraState[player] = nil
    end
    destroyAuraInstances(player.Character)
end)
    function AM.Destroy()
        if managerDestroyed then return end
        managerDestroyed = true
        if playerRemovingConnection then pcall(function() playerRemovingConnection:Disconnect() end); playerRemovingConnection = nil end
        AM.ClearAllAuras()
        AM.CleanupShaders()
    end
    return AM
end)()

        self._frames = 0
        self._elapsed = 0

            end

            self._connection = nil

        self._opts = table.clone(opts or {})

        for i, option in ipairs(self._opts) do
            local button = create("TextButton", {
                Name = "O_" .. i,
                LayoutOrder = i,
                Size = UDim2.new(1, -2, 0, 48),
                BackgroundColor3 = THEME.Element,
                BackgroundTransparency = 0.14,
                Text = "   " .. tostring(option),
                TextColor3 = Color3.fromRGB(225, 231, 248),
                TextTransparency = 0,
                Font = Enum.Font.GothamSemibold,
                TextSize = 15,
                TextXAlignment = Enum.TextXAlignment.Left,
                AutoButtonColor = false,
                ZIndex = 502,
                Parent = self.Scroll
            })
            create("UICorner", {CornerRadius = UDim.new(0, 8), Parent = button})
            create("UIStroke", {
                Color = Theme.Stroke,
                Thickness = 1,
                Transparency = 0.68,
                Parent = button
            })
            create("UIPadding", {PaddingLeft = UDim.new(0, 10), Parent = button})

            button.MouseEnter:Connect(function()
                if self.Value ~= option then
                    button.BackgroundTransparency = 0.04
                end
            end)
            button.MouseLeave:Connect(function()
                if self.Value ~= option then
                    -- Keep idle rows opaque; the old transparency=1 made them disappear.
                    button.BackgroundTransparency = 0.14
                end
            end)
            button.Activated:Connect(function()
                self:_select(i)
                self:Close()
                self:_fire(self.Value)
            end)
            self._btns[i] = button
        end

        self._selIdx = (keep ~= false and self.Value ~= nil) and table.find(self._opts, self.Value) or nil
        if not self._selIdx then
            self.Value = nil
            self.Header.Text = "Select..."
        end
        if self.Value then self.Header.Text = tostring(self.Value) end

        local contentHeight = #self._opts * 48 + 18
        self.Scroll.CanvasSize = UDim2.fromOffset(0, contentHeight)
        self._targetHeight = math.min(contentHeight, UserInputService.TouchEnabled and 230 or 220)
        if self._selIdx then self:_select(self._selIdx) end
    end

    function D:Open()
        if self.IsOpen or self._destroyed or not self.Frame.Parent then return end
        if activeDropdown and activeDropdown ~= self then
            pcall(function() activeDropdown:Close() end)
        end
        if self._tween then
            pcall(function() self._tween:Cancel() end)
            self._tween = nil
        end

        activeDropdown = self
        self.IsOpen = true
        self:_enforceStyle()
        if self.Chevron then self.Chevron.Text = "⌃" end

        local camera = workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or Vector2.new(800, 600)
        local topInset, bottomInset = Vector2.zero, Vector2.zero
        pcall(function()
            topInset, bottomInset = GuiService:GetGuiInset()
        end)

        local rowPosition = self.Frame.AbsolutePosition
        local rowSize = self.Frame.AbsoluteSize
        local rowTop = rowPosition.Y
        local rowBottom = rowTop + rowSize.Y
        -- ScreenGui.IgnoreGuiInset=false uses the Core UI safe-area origin already.
        local safeTop = 6
        local safeBottom = math.max(safeTop + 72, viewport.Y - topInset.Y - bottomInset.Y - 6)
        local availableAbove = math.max(0, rowTop - safeTop - 4)
        local availableBelow = math.max(0, safeBottom - rowBottom - 4)
        local desired = math.max(60, self._targetHeight)

        self._openAbove = availableBelow < math.min(desired, 112) and availableAbove > availableBelow
        local available = self._openAbove and availableAbove or availableBelow
        self._displayHeight = math.min(desired, math.max(60, available))
        self._displayHeight = math.min(self._displayHeight, math.max(60, safeBottom - safeTop - 12))

        local width = math.max(160, math.min(rowSize.X, viewport.X - 12))
        local x = math.clamp(rowPosition.X, 6, math.max(6, viewport.X - width - 6))
        local y = self._openAbove and (rowTop - self._displayHeight - 4) or (rowBottom + 4)
        y = math.clamp(y, safeTop, math.max(safeTop, safeBottom - self._displayHeight))

        self.List.Position = UDim2.fromOffset(x, y)
        self.Scroll.CanvasPosition = Vector2.zero
        self.List.Size = UDim2.fromOffset(width, 0)
        self.List.Visible = true

        self._tween = TweenService:Create(self.List, OP, {
            Size = UDim2.fromOffset(width, self._displayHeight)
        })
        self._tween:Play()
    end

    function D:Close()
        if not self.IsOpen or self._destroyed then return end
        self.IsOpen = false
        if activeDropdown == self then activeDropdown = nil end
        if self.Chevron then self.Chevron.Text = "⌄" end
        if self._tween then
            pcall(function() self._tween:Cancel() end)
            self._tween = nil
        end
        self._tween = TweenService:Create(self.List, OP, {
            Size = UDim2.fromOffset(self.List.AbsoluteSize.X, 0)
        })
        self._tween.Completed:Once(function(state)
            if state == Enum.PlaybackState.Completed and not self.IsOpen and self.List and self.List.Parent then
                self.List.Visible = false
            end
        end)
        self._tween:Play()
    end

    function D:Destroy()
        if self._destroyed then return end
        self._destroyed = true
        self.IsOpen = false
        if activeDropdown == self then activeDropdown = nil end
        if self._tween then
            pcall(function() self._tween:Cancel() end)
            self._tween = nil
        end
        pcall(function() self.List:Destroy() end)
        pcall(function() self.Frame:Destroy() end)
        table.clear(self._btns)
    end

    function D.CloseAll()
        if activeDropdown then
            pcall(function() activeDropdown:Close() end)
        end
    end

    function D.DestroyOverlay()
        activeDropdown = nil
        if popupGui then
            pcall(function() popupGui:Destroy() end)
            popupGui = nil
        end
    end

    return D
end)()

LootTracker = (function()
    local LT = {}; LT.__index = LT
    local UPDATE_INTERVAL, SCAN_BATCH_SIZE, MAX_TRACKED = 0.2, 500, 150
    local STUDS_PER_METER = 3.571
    -- Loot ESP is intentionally candy-only. AutoFarm keeps its separate coin target logic.
    local TRACKED = {
        Candy = Color3.fromRGB(255, 110, 220)
    }
    local function getTrackedColor(inst)
        if not inst or not (inst:IsA("BasePart") or inst:IsA("Model")) then return nil end
        local n = string.lower(inst.Name)
        -- Never highlight container/group models, even if their names contain "candy".
        if n:find("container", 1, true) or n == "loot" or n == "collectibles" then return nil end
        -- Accept candy-named pickups only; coins, gems, cash, weapons and knives are excluded.
        if n == "candy" or n == "candies" or n:find("candy", 1, true) then
            return TRACKED.Candy
        end
        return nil
    end
    local function hasTrackedAncestor(inst)
        local ancestor = inst.Parent
        while ancestor and ancestor ~= Workspace do
            if getTrackedColor(ancestor) then return true end
            ancestor = ancestor.Parent
        end
        return false
    end
    local function getPosition(item)
        local part = getTouchPart(item)
        return part and part.Parent and part.Position or nil
    end
    function LT.new(parentGui)
        local self = setmetatable({}, LT)
        self._parentGui, self._running, self._destroyed = parentGui, false, false
        self._cache, self._count, self._connections, self._scanToken = {}, 0, {}, 0
        self._pending, self._promoting = {}, false
        self._highlightFolder = Instance.new("Folder")
        self._highlightFolder.Name = "LootTrackerHighlights"
        self._highlightFolder.Parent = Workspace
        return self
    end
    function LT:_track(item)
        if self._cache[item] or self._destroyed or not item or not item.Parent or hasTrackedAncestor(item) then return end
        local color = getTrackedColor(item)
        if not color or not getTouchPart(item) then return end
        if self._count >= MAX_TRACKED then self._pending[item] = true; return end
        self._pending[item] = nil
        local highlight, billboard
        local ok, entry = pcall(function()
            highlight = Instance.new("Highlight")
            highlight.Name = "LootHighlight"
            highlight.Adornee = item
            highlight.FillColor = color
            highlight.FillTransparency = 0.62
            highlight.OutlineColor = color
            highlight.OutlineTransparency = 0
            highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            highlight.Parent = self._highlightFolder

            billboard = Instance.new("BillboardGui")
            billboard.Name = "LootBillboard"
            billboard.Adornee = getTouchPart(item)
            billboard.AlwaysOnTop = true
            billboard.Size = UDim2.new(0, 132, 0, 38)
            billboard.StudsOffsetWorldSpace = Vector3.new(0, 2.7, 0)
            billboard.MaxDistance = 300
            billboard.ResetOnSpawn = false
            billboard.Active = false
            billboard.Parent = self._parentGui
            local textLabel = Instance.new("TextLabel")
            textLabel.Size = UDim2.new(1, 0, 1, 0)
            textLabel.BackgroundColor3 = THEME.Surface
            textLabel.BackgroundTransparency = 0.16
            textLabel.BorderSizePixel = 0
            textLabel.Active = false
            textLabel.Text = item.Name
            textLabel.TextColor3 = color
            textLabel.TextStrokeTransparency = 1
            textLabel.Font = Enum.Font.GothamBold
            textLabel.TextSize = 12
            textLabel.TextWrapped = true
            textLabel.Parent = billboard
            create("UICorner", {CornerRadius = UDim.new(0, 8), Parent = textLabel})
            create("UIStroke", {Color = color, Thickness = 1, Transparency = 0.28, Parent = textLabel})
            create("UIGradient", {Color = ColorSequence.new(Color3.fromRGB(28, 22, 44), Color3.fromRGB(13, 19, 32)), Rotation = 0, Parent = textLabel})
            create("UIPadding", {PaddingLeft = UDim.new(0, 5), PaddingRight = UDim.new(0, 5), Parent = textLabel})
            return { highlight = highlight, billboard = billboard, label = textLabel, name = item.Name, lastText = "" }
        end)
        if ok and entry then
            self._cache[item] = entry
            self._count += 1
        else
            if highlight then pcall(function() highlight:Destroy() end) end
            if billboard then pcall(function() billboard:Destroy() end) end
        end
    end
    function LT:_promotePending()
        if self._promoting or not self._running or self._destroyed or self._count >= MAX_TRACKED then return end
        self._promoting = true
        task.defer(function()
            self._promoting = false
            if not self._running or self._destroyed then return end
            for item in pairs(self._pending) do
                if not item or not item.Parent or not item:IsDescendantOf(Workspace) then
                    self._pending[item] = nil
                elseif self._count < MAX_TRACKED then
                    self._pending[item] = nil
                    self:_track(item)
                else
                    break
                end
            end
        end)
    end
    function LT:_untrack(item)
        local entry = self._cache[item]
        if not entry then self._pending[item] = nil; return end
        self._cache[item] = nil
        self._count = math.max(0, self._count - 1)
        pcall(function() entry.highlight:Destroy() end)
        pcall(function() entry.billboard:Destroy() end)
        self:_promotePending()
    end
    function LT:_onRemoving(inst)
        self._pending[inst] = nil
        if self._cache[inst] then self:_untrack(inst); return end
        if inst:IsA("Model") or inst:IsA("Folder") then
            local remove = {}
            for item in pairs(self._cache) do
                if item:IsDescendantOf(inst) then table.insert(remove, item) end
            end
            for item in pairs(self._pending) do
                if item == inst or item:IsDescendantOf(inst) then self._pending[item] = nil end
            end
            for _, item in ipairs(remove) do self:_untrack(item) end
        end
    end
    function LT:_refresh()
        local player = Players.LocalPlayer
        local character = player and player.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        for item, entry in pairs(self._cache) do
            if not item.Parent or not item:IsDescendantOf(Workspace) then
                self:_untrack(item)
            else
                local position = getPosition(item)
                local part = getTouchPart(item)
                if position and part and part.Parent then
                    if entry.billboard.Adornee ~= part then entry.billboard.Adornee = part end
                    local text = string.format("%s\n%.0f m", entry.name, (position - root.Position).Magnitude / STUDS_PER_METER)
                    if text ~= entry.lastText then entry.lastText = text; entry.label.Text = text end
                end
            end
        end
    end
    function LT:Start()
        if self._running or self._destroyed then return end
        self._running = true
        self._scanToken += 1
        local token = self._scanToken
        table.insert(self._connections, Workspace.DescendantAdded:Connect(function(inst)
            task.defer(function()
                if not self._running or self._scanToken ~= token then return end
                local current = inst
                while current and current ~= Workspace do
                    if getTrackedColor(current) then self:_track(current) end
                    current = current.Parent
                end
            end)
        end))
        table.insert(self._connections, Workspace.DescendantRemoving:Connect(function(inst) self:_onRemoving(inst) end))
        local elapsed = 0
        table.insert(self._connections, RunService.Heartbeat:Connect(function(dt)
            elapsed += dt
            if elapsed < UPDATE_INTERVAL then return end
            elapsed = 0
            pcall(function() self:_refresh() end)
        end))
        task.spawn(function()
            local processed = 0
            for _, inst in ipairs(Workspace:GetDescendants()) do
                if not self._running or self._scanToken ~= token then return end
                if getTrackedColor(inst) then self:_track(inst) end
                processed += 1
                if processed % SCAN_BATCH_SIZE == 0 then task.wait() end
            end
        end)
    end
    function LT:Stop()
        if not self._running then return end
        self._running = false
        self._scanToken += 1
        for _, conn in ipairs(self._connections) do pcall(function() conn:Disconnect() end) end
        table.clear(self._connections)
        local items = {}
        for item in pairs(self._cache) do table.insert(items, item) end
        for _, item in ipairs(items) do self:_untrack(item) end
        table.clear(self._pending)
    end
    function LT:Destroy()
        if self._destroyed then return end
        self:Stop()
        self._destroyed = true
        pcall(function() self._highlightFolder:Destroy() end)
    end
    return LT
end)()

local function getActiveMap()
    local spawnFallback = nil
    local bestMap, bestScore = nil, -1
    for _, candidate in ipairs(Workspace:GetChildren()) do
        if candidate:IsA("Model") or candidate:IsA("Folder") then
            local container = candidate:FindFirstChild("CoinContainer") or candidate:FindFirstChild("CoinContainer", true)
            if container then
                local score = 0
                for _, item in ipairs(container:GetDescendants()) do
                    if item:IsA("BasePart") and item.Transparency < 0.98
                        and item:GetAttribute("Collected") ~= true then
                        local touch = item:FindFirstChild("TouchInterest")
                            or item:FindFirstChildWhichIsA("TouchTransmitter", true)
                        if touch then score += 1 end
                    end
                end
                local name = string.lower(candidate.Name)
                if name == "map" or name:find("map", 1, true) then score += 0.01 end
                if score > bestScore then
                    bestMap, bestScore = candidate, score
                end
            elseif not spawnFallback and candidate:FindFirstChild("Spawns") then
                spawnFallback = candidate
            end
        end
    end
    return bestMap or spawnFallback
end

local function getActiveCoinContainer()
    local map = getActiveMap()
    return map and (map:FindFirstChild("CoinContainer") or map:FindFirstChild("CoinContainer", true)) or nil
end

weaponTracker = (function()
    local API = {}
    local skipped = setmetatable({}, { __mode = "k" })
    local drops = setmetatable({}, { __mode = "k" })
    local running = false
    local addedConnection = nil
    local lastScanAt = 0

    local function remember(inst)
        if inst and inst:IsA("BasePart") and string.lower(inst.Name) == "gundrop" then
            drops[inst] = true
        end
    end

    local function scan()
        lastScanAt = os.clock()
        for _, inst in ipairs(Workspace:GetDescendants()) do remember(inst) end
    end

    local function findTouchPart(drop)
        if not drop or not drop.Parent or not drop:IsDescendantOf(Workspace) then return nil end
        local transmitter = drop:FindFirstChildWhichIsA("TouchTransmitter", true)
            or drop:FindFirstChild("TouchInterest", true)
        local hitbox = transmitter and transmitter.Parent
        if hitbox and hitbox:IsA("BasePart") and hitbox.Parent then return hitbox end

        -- PulseHub's GunDrop lookup accepts the named physical drop itself when a
        -- TouchTransmitter is not exposed by the executor/client build.
        if drop:IsA("BasePart") and drop.Size.Magnitude > 0 and drop.Transparency < 0.99 then
            return drop
        end

        local best, bestScore = nil, -math.huge
        for _, child in ipairs(drop:GetDescendants()) do
            if child:IsA("BasePart") and child.Size.Magnitude > 0 then
                local score = (child.CanTouch and 8 or 0) + (child.Transparency < 0.99 and 2 or 0)
                if string.lower(child.Name):find("hit", 1, true) then score += 4 end
                if score > bestScore then best, bestScore = child, score end
            end
        end
        return best
    end

    local function findDroppedGun()
        if not running then return nil, nil end
        if os.clock() - lastScanAt > 1.25 then scan() end
        local activeMap = getActiveMap()
        local bestDrop, bestPart, bestScore = nil, nil, -math.huge
        for drop in pairs(drops) do
            if not drop or not drop.Parent or not drop:IsDescendantOf(Workspace)
                or string.lower(drop.Name) ~= "gundrop" then
                drops[drop] = nil
            else
                local part = findTouchPart(drop)
                if part then
                    local score = 0
                    if activeMap and drop:IsDescendantOf(activeMap) then score += 100 end
                    if part.CanTouch then score += 10 end
                    if part.Transparency < 0.99 then score += 2 end
                    if score > bestScore then
                        bestDrop, bestPart, bestScore = drop, part, score
                    end
                end
            end
        end
        return bestDrop, bestPart
    end

    function API:Start()
        if running then return end
        running = true
        table.clear(skipped)
        table.clear(drops)
        scan()
        addedConnection = Workspace.DescendantAdded:Connect(remember)
    end

    function API:Stop()
        running = false
        if addedConnection then pcall(function() addedConnection:Disconnect() end); addedConnection = nil end
        table.clear(skipped)
        table.clear(drops)
        lastScanAt = 0
    end

    function API:IsValid(obj)
        if not running or not obj then return false end
        local drop, part = findDroppedGun()
        return drop == obj and part ~= nil
    end

    function API:GetTouchPart(obj)
        if not running or not obj then return nil end
        local drop, part = findDroppedGun()
        if drop == obj then return part end
        return nil
    end

    function API:GetClosest(origin, maxDist)
        if not running or not origin then return nil, nil end
        local drop, part = findDroppedGun()
        if not drop or not part then return nil, nil end
        local now = os.clock()
        if (skipped[drop] or 0) > now then return nil, nil end
        local distance = (part.Position - origin).Magnitude
        if maxDist and distance > maxDist then return nil, distance end
        return drop, distance
    end

    function API:GetStatus(origin, maxDist)
        if not running then return "OFF" end
        local drop, part = findDroppedGun()
        if not drop or not part then return "NO DROP" end
        if origin and maxDist and (part.Position - origin).Magnitude > maxDist then return "OUT OF RANGE" end
        return "READY"
    end

    function API:Skip(obj, seconds)
        if not running or not obj then return end
        skipped[obj] = os.clock() + math.clamp(tonumber(seconds) or 2, 0.25, 15)
    end

    function API:Destroy()
        self:Stop()
    end

    return API
end)()

local tracerToolConnections = setmetatable({}, { __mode = "k" })
local tracerWatchConnections = {}
local function startTracerHooks()
    if #tracerWatchConnections > 0 then return end

    local watchedContainers = setmetatable({}, { __mode = "k" })

    local function bindTool(tool)
        if not tool or not tool:IsA("Tool") or tracerToolConnections[tool] then return end
        local name = string.lower(tool.Name)
        if not (name:find("gun", 1, true) or name:find("pistol", 1, true) or name:find("revolver", 1, true)) then return end

        local activatedConnection = tool.Activated:Connect(function()
            if not CFG.Tracers or not isRunning then return end
            local camera = Workspace.CurrentCamera
            if not camera then return end

            local viewport = camera.ViewportSize
            local ray = camera:ViewportPointToRay(viewport.X * 0.5, viewport.Y * 0.5)
            local params = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            local excluded = {}
            if LocalPlayer.Character then table.insert(excluded, LocalPlayer.Character) end
            if aimMarker then table.insert(excluded, aimMarker) end
            params.FilterDescendantsInstances = excluded

            local hit = Workspace:Raycast(ray.Origin, ray.Direction * 1000, params)
            local endpoint = hit and hit.Position or (ray.Origin + ray.Direction * 250)
            local handle = tool:FindFirstChild("Handle")
            local origin = handle and handle:IsA("BasePart") and handle.Position or ray.Origin
            TracerEffect.Create(origin, endpoint, colorFromName and colorFromName(CFG.TracerColor) or Color3.fromRGB(255,220,48))
        end)

        local ancestryConnection
        ancestryConnection = tool.AncestryChanged:Connect(function(_, parent)
            if parent ~= nil then return end
            pcall(function() activatedConnection:Disconnect() end)
            if ancestryConnection then pcall(function() ancestryConnection:Disconnect() end) end
            tracerToolConnections[tool] = nil
        end)
        tracerToolConnections[tool] = {
            Activated = activatedConnection,
            Ancestry = ancestryConnection
        }
    end

    local function trackConnection(connection)
        table.insert(tracerWatchConnections, connection)
        return connection
    end

    local function watch(container)
        if not container or watchedContainers[container] then return end
        watchedContainers[container] = true

        for _, child in ipairs(container:GetChildren()) do bindTool(child) end

        local added
        local removing
        local function detach()
            if not watchedContainers[container] then return end
            watchedContainers[container] = nil

            if added then pcall(function() added:Disconnect() end) end
            if removing then pcall(function() removing:Disconnect() end) end

            for index = #tracerWatchConnections, 1, -1 do
                local connection = tracerWatchConnections[index]
                if connection == added or connection == removing then
                    table.remove(tracerWatchConnections, index)
                end
            end
        end

        added = trackConnection(container.ChildAdded:Connect(bindTool))
        removing = trackConnection(container.AncestryChanged:Connect(function(_, parent)
            if parent == nil then detach() end
        end))
    end

    local function watchCurrentBackpack()
        local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
        if backpack then watch(backpack) end
    end

    watchCurrentBackpack()
    if LocalPlayer.Character then watch(LocalPlayer.Character) end

    trackConnection(LocalPlayer.ChildAdded:Connect(function(child)
        if child:IsA("Backpack") then watch(child) end
    end))
    trackConnection(LocalPlayer.CharacterAdded:Connect(function(character)
        watch(character)
        watchCurrentBackpack()
    end))
end

stopTracerHooks = function()
    for _, connection in ipairs(tracerWatchConnections) do
        pcall(function() connection:Disconnect() end)
    end
    table.clear(tracerWatchConnections)

    -- Disconnect both the firing callback and the destruction watcher for every tool.
    for tool, connections in pairs(tracerToolConnections) do
        if connections then
            if connections.Activated then pcall(function() connections.Activated:Disconnect() end) end
            if connections.Ancestry then pcall(function() connections.Ancestry:Disconnect() end) end
        end
        tracerToolConnections[tool] = nil
    end

    if TracerEffect then pcall(function() TracerEffect.DestroyAll() end) end
end

SkeletonVisualizer = (function() local SV = {} SV.__index = SV local MAX_PLAYERS = 20 local DEFAULT_COLOR = Color3.fromRGB(255, 255, 255) local BEAM_WIDTH = 0.12 local BEAM_ZOFFSET = 1 local PART_WAIT = 5 local FOLDER_NAME = "SkeletonVisualizer" local JOINTS = { Head = { R15 = "Head", R6 = "Head" }, UpperTorso = { R15 = "UpperTorso", R6 = "Torso" }, LowerTorso = { R15 = "LowerTorso", R6 = "Torso" }, LeftArm = { R15 = "LeftUpperArm", R6 = "Left Arm" }, RightArm = { R15 = "RightUpperArm", R6 = "Right Arm" }, LeftLeg = { R15 = "LeftUpperLeg", R6 = "Left Leg" }, RightLeg = { R15 = "RightUpperLeg", R6 = "Right Leg" } } local BONES = { { "Head", "UpperTorso" }, { "UpperTorso", "LowerTorso" }, { "UpperTorso", "LeftArm" }, { "UpperTorso", "RightArm" }, { "LowerTorso", "LeftLeg" }, { "LowerTorso", "RightLeg" } } function SV.new(parentGui) local self = setmetatable({}, SV) self._destroyed = false self._visible = false self._color = DEFAULT_COLOR self._players = {} self._count = 0 self._connections = {} local localPlayer = Players.LocalPlayer for _, player in ipairs(Players:GetPlayers()) do if player ~= localPlayer then self:_addPlayer(player) end end table.insert(self._connections, Players.PlayerAdded:Connect(function(player) self:_addPlayer(player) end)) table.insert(self._connections, Players.PlayerRemoving:Connect(function(player) self:_removePlayer(player) self:_fillSlots() end)) return self end function SV:_clear(player) local state = self._players[player] if not state then return end if state.folder then pcall(function() state.folder:Destroy() end) state.folder = nil end for _, attachment in ipairs(state.attachments) do pcall(function() attachment:Destroy() end) end table.clear(state.attachments) table.clear(state.beams) end function SV:_build(player, character)
    local state = self._players[player]
    if not state or self._destroyed or not self._visible then return end

    state.token += 1
    local token = state.token
    self:_clear(player)

    local function stale()
        return self._destroyed
            or self._players[player] ~= state
            or state.token ~= token
            or player.Character ~= character
            or not character.Parent
            or not self._visible
    end

    local ok, err = pcall(function()
        local humanoid = character:WaitForChild("Humanoid", PART_WAIT)
        if not humanoid or stale() then return end

        local rig = humanoid.RigType == Enum.HumanoidRigType.R15 and "R15" or "R6"
        local partByJoint = {}

        for jointName, rigNames in pairs(JOINTS) do
            local part = character:WaitForChild(rigNames[rig], PART_WAIT)
            if stale() then return end
            if part and part:IsA("BasePart") then
                partByJoint[jointName] = part
            end
        end

        local motors = {}
        for _, descendant in ipairs(character:GetDescendants()) do
            if descendant:IsA("Motor6D") and descendant.Part0 and descendant.Part1 then
                table.insert(motors, descendant)
            end
        end

        local function edgeOffset(part, worldDirection)
            local localDirection = part.CFrame:VectorToObjectSpace(worldDirection)
            local half = part.Size * 0.5
            local ratio = math.max(
                math.abs(localDirection.X) / math.max(half.X, 0.05),
                math.abs(localDirection.Y) / math.max(half.Y, 0.05),
                math.abs(localDirection.Z) / math.max(half.Z, 0.05)
            )
            if ratio <= 0 then return Vector3.zero end
            return localDirection / ratio
        end

        local function getJointOffsets(partA, partB)
            for _, motor in ipairs(motors) do
                if motor.Part0 == partA and motor.Part1 == partB then
                    return motor.C0.Position, motor.C1.Position
                elseif motor.Part0 == partB and motor.Part1 == partA then
                    return motor.C1.Position, motor.C0.Position
                end
            end

            -- Fallback for custom rigs without a direct Motor6D between the expected parts.
            local delta = partB.Position - partA.Position
            if delta.Magnitude < 0.001 then
                return Vector3.zero, Vector3.zero
            end
            local direction = delta.Unit
            return edgeOffset(partA, direction), edgeOffset(partB, -direction)
        end

        local folder = Instance.new("Folder")
        folder.Name = FOLDER_NAME

        for _, bone in ipairs(BONES) do
            if stale() then
                folder:Destroy()
                if self._players[player] == state and state.token == token then
                    self:_clear(player)
                end
                return
            end

            local nameA, nameB = bone[1], bone[2]
            local partA, partB = partByJoint[nameA], partByJoint[nameB]

            if partA and partB and partA ~= partB then
                local offsetA, offsetB = getJointOffsets(partA, partB)

                local attachmentA = Instance.new("Attachment")
                attachmentA.Name = "SkeletonAttachment_" .. nameA .. "_" .. nameB .. "_A"
                attachmentA.Position = offsetA
                attachmentA.Parent = partA
                table.insert(state.attachments, attachmentA)

                local attachmentB = Instance.new("Attachment")
                attachmentB.Name = "SkeletonAttachment_" .. nameA .. "_" .. nameB .. "_B"
                attachmentB.Position = offsetB
                attachmentB.Parent = partB
                table.insert(state.attachments, attachmentB)

                local beam = Instance.new("Beam")
                beam.Name = nameA .. "_" .. nameB
                beam.Attachment0 = attachmentA
                beam.Attachment1 = attachmentB
                beam.Color = ColorSequence.new(self._color)
                beam.Width0 = BEAM_WIDTH
                beam.Width1 = BEAM_WIDTH
                beam.FaceCamera = true
                beam.LightEmission = 1
                beam.LightInfluence = 0
                beam.Transparency = NumberSequence.new(0)
                beam.Segments = 1
                beam.ZOffset = BEAM_ZOFFSET
                beam.Enabled = self._visible
                beam.Parent = folder
                table.insert(state.beams, beam)
            end
        end

        if stale() then
            folder:Destroy()
            if self._players[player] == state and state.token == token then
                self:_clear(player)
            end
            return
        end

        folder.Parent = character
        state.folder = folder
    end)

    if not ok then
        warn("[SkeletonVisualizer] build failed for " .. player.Name .. ": " .. tostring(err))
        if self._players[player] == state and state.token == token then
            self:_clear(player)
        end
    end
end
function SV:_addPlayer(player) if self._destroyed or player == Players.LocalPlayer then return end if self._players[player] then return end if self._count >= MAX_PLAYERS then return end local state = { token = 0, folder = nil, attachments = {}, beams = {}, connections = {} } self._players[player] = state self._count += 1 table.insert(state.connections, player.CharacterAdded:Connect(function(character) task.spawn(function() self:_build(player, character) end) end)) table.insert(state.connections, player.CharacterRemoving:Connect(function() state.token += 1 self:_clear(player) end)) if player.Character then task.spawn(function() self:_build(player, player.Character) end) end end function SV:_removePlayer(player) local state = self._players[player] if not state then return end state.token += 1 for _, conn in ipairs(state.connections) do pcall(function() conn:Disconnect() end) end self:_clear(player) self._players[player] = nil self._count -= 1 end function SV:_fillSlots() if self._destroyed then return end for _, player in ipairs(Players:GetPlayers()) do if self._count >= MAX_PLAYERS then break end self:_addPlayer(player) end end function SV:SetVisible(visible)
    if self._destroyed then return end
    visible = visible == true
    if self._visible == visible then return end
    self._visible = visible
    if not visible then
        for player, state in pairs(self._players) do
            state.token += 1
            self:_clear(player)
        end
        return
    end
    for player in pairs(self._players) do
        local character = player.Character

getTouchPart = function(obj)
    if not obj or not obj.Parent then return nil end
    if obj:IsA("BasePart") then return obj end
    if not obj:IsA("Model") then return nil end
    local best, bestScore = nil, -math.huge
    for _, descendant in ipairs(obj:GetDescendants()) do
        if descendant:IsA("BasePart") then
            local score = 0
            if descendant.CanTouch then score += 8 end
            if descendant.CanCollide then score += 2 end
            if string.lower(descendant.Name):find("coin", 1, true) then score += 4 end
            if string.lower(descendant.Name):find("hit", 1, true) or string.lower(descendant.Name):find("touch", 1, true) then score += 3 end
            if descendant.Transparency < 1 then score += 1 end
            if score > bestScore then best, bestScore = descendant, score end
        end
    end
    return best
end

local function isCoinContainerName(name)
    return name == "coincontainer" or name == "coin_container" or name == "coins"
        or name == "coinfolder" or name == "coin_folder" or name == "coinspawns"
        or name == "coin_spawn" or name == "coinspawn"
end

local function getCoinContainer(inst)
    local container = getActiveCoinContainer()
    if not container or not inst or not inst.Parent then return nil end
    if inst == container or inst:IsDescendantOf(container) then return container end
    return nil
end

local function hasCoinAncestor(inst)
    local ancestor = inst and inst.Parent
    while ancestor and ancestor ~= Workspace do
        local name = string.lower(ancestor.Name)
        if not isCoinContainerName(name)
            and (ancestor:IsA("BasePart") or ancestor:IsA("Model"))
            and name:match("^coin") then
            return true
        end
        ancestor = ancestor.Parent
    end
    return false
end


-- MM2 round coins are BaseParts with a TouchInterest/TouchTransmitter.
-- Keep the predicate strict so coin containers, map props and decorative loot are never farm targets.
local function isCoin(inst)
    if not inst or not inst:IsA("BasePart") or not inst.Parent then return false end
    if inst:GetAttribute("Collected") == true then return false end
    local touch = inst:FindFirstChild("TouchInterest")
        or inst:FindFirstChildWhichIsA("TouchTransmitter", true)
    if not touch then return false end
    local coinId = inst:GetAttribute("CoinID")
    if coinId ~= nil and farmBagEventFull[coinId] then return false end
    return true
end

-- Return the floor raycast below a coin, excluding the avatar and coin holder.
-- coinTracker caches this result so the farm does not raycast every frame.
local function hasGroundBelow(obj, part)
    if not part or not part.Parent then return nil end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.IgnoreWater = true
    pcall(function() params.RespectCanCollide = true end)
    local ignored = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character then table.insert(ignored, player.Character) end
    end
    if obj then table.insert(ignored, obj) end
    local container = getActiveCoinContainer()
    if container then table.insert(ignored, container) end
    params.FilterDescendantsInstances = ignored
    local ok, result = pcall(function()
        return Workspace:Raycast(part.Position + Vector3.new(0, 0.75, 0), Vector3.new(0, -60, 0), params)
    end)
    return ok and result or nil
end


coinTracker = (function()
    local API = {}
    local coins = {}
    local observedAt = setmetatable({}, { __mode = "k" })
    local skipped = setmetatable({}, { __mode = "k" })
    local failures = setmetatable({}, { __mode = "k" })
    local groundCache = setmetatable({}, { __mode = "k" })
    local GROUND_CACHE_TIME = 0.6
    local GROUND_CACHE_MOVE = 1.5
    local running = false
    local scanToken = 0
    local boundContainer = nil
    local lastPeriodicScanAt = 0
    local containerAdded = nil
    local containerRemoving = nil
    local mapAdded = nil
    local mapDescendantAdded = nil
    local mapRemoving = nil

    local function disconnect(connection)
        if connection then pcall(function() connection:Disconnect() end) end
    end

    local function clearTargets()
        table.clear(coins)
        table.clear(skipped)
        table.clear(failures)
        table.clear(observedAt)
        table.clear(groundCache)
    end

    local function onRemoving(inst)
        local index = table.find(coins, inst)
        if index then table.remove(coins, index) end
        if inst then
            skipped[inst] = nil
            failures[inst] = nil
            observedAt[inst] = nil
            groundCache[inst] = nil
        end
    end

    local function getCachedGround(obj, part)
        local now = os.clock()
        local cached = groundCache[obj]
        if cached and cached.part == part and now - cached.checkedAt < GROUND_CACHE_TIME
            and (part.Position - cached.position).Magnitude < GROUND_CACHE_MOVE then
            return cached.result
        end
        local result = hasGroundBelow(obj, part)
        -- Like PulseHub's coinInfo fallback, do not discard a touchable coin merely
        -- because a non-collidable/streamed floor produced no raycast result.
        if not result then
            result = {
                Position = Vector3.new(part.Position.X, part.Position.Y - 3, part.Position.Z),
                _liberiumEstimated = true
            }
        end
        groundCache[obj] = {
            part = part,
            position = part.Position,
            checkedAt = now,
            result = result
        }
        return result
    end

    local function add(inst)
        local current = inst
        while current and current ~= Workspace and current ~= boundContainer do
            if isCoin(current) and not table.find(coins, current) then
                table.insert(coins, current)
                observedAt[current] = os.clock()
            end
            current = current.Parent
        end
    end

    local function unbindContainer()
        disconnect(containerAdded)
        disconnect(containerRemoving)
        containerAdded = nil
        containerRemoving = nil
        boundContainer = nil
        clearTargets()
    end

    local function bindContainer()
        if not running then return end
        local container = getActiveCoinContainer()
        if container == boundContainer then return end
        unbindContainer()
        if not container or not container.Parent or not container:IsDescendantOf(Workspace) then return end

        boundContainer = container
        containerAdded = container.DescendantAdded:Connect(function(inst)
            if running and boundContainer == container then add(inst) end
        end)
        containerRemoving = container.DescendantRemoving:Connect(function(inst)
            if boundContainer == container then onRemoving(inst) end
        end)

        local token = scanToken
        task.spawn(function()
            local descendants = container:GetDescendants()
            for index, inst in ipairs(descendants) do
                if not running or scanToken ~= token or boundContainer ~= container then return end
                add(inst)
                if index % 120 == 0 then task.wait() end
            end
        end)
    end

    function API:Start()
        if running then return end
        running = true
        scanToken += 1
        local token = scanToken

        mapAdded = Workspace.ChildAdded:Connect(function()
            if running and scanToken == token then task.defer(bindContainer) end
        end)
        mapDescendantAdded = Workspace.DescendantAdded:Connect(function(inst)
            if not running or scanToken ~= token then return end
            if string.lower(inst.Name) == "coincontainer" then
                task.defer(bindContainer)
            end
        end)
        mapRemoving = Workspace.DescendantRemoving:Connect(function(inst)
            if not running or scanToken ~= token then return end
            if boundContainer and (inst == boundContainer or boundContainer:IsDescendantOf(inst)) then
                task.defer(bindContainer)
            end
        end)

        bindContainer()
    end

    function API:Stop()
        if not running then return end
        running = false
        scanToken += 1
        disconnect(mapAdded)
        disconnect(mapDescendantAdded)
        disconnect(mapRemoving)
        mapAdded = nil
        mapDescendantAdded = nil
        mapRemoving = nil
        unbindContainer()
    end

    function API:IsValid(obj)
        if not running or not obj or not obj.Parent or not obj:IsDescendantOf(Workspace) then return false end
        if not boundContainer or not obj:IsDescendantOf(boundContainer) or not isCoin(obj) then return false end
        local part = getTouchPart(obj)
        return part ~= nil and part.Parent ~= nil and part.Transparency < 0.98
    end

    function API:GetGround(obj, part)
        if not running or not obj or not obj.Parent then return nil end
        part = part or getTouchPart(obj)
        if not part or not part.Parent then return nil end
        return getCachedGround(obj, part)
    end

    function API:GetClosest(origin, maxDist, method, murdererPosition, lastCollectedPosition)
        if not running or not origin then return nil, nil end
        if getActiveCoinContainer() ~= boundContainer then bindContainer() end
        if not boundContainer then return nil, nil end

        method = method or CFG.FarmMethod or "Fast"
        local now = os.clock()
        if now - lastPeriodicScanAt >= 0.75 then
            lastPeriodicScanAt = now
            -- DescendantAdded can fire before MM2 attaches the TouchInterest.
            -- Periodically retry direct children so late-initialized coins enter the cache.
            for _, child in ipairs(boundContainer:GetChildren()) do add(child) end
        end
        local candidates = {}
        for i = #coins, 1, -1 do
            local obj = coins[i]
            if not obj or not obj.Parent or not obj:IsDescendantOf(boundContainer) or not isCoin(obj) then
                table.remove(coins, i)
                if obj then skipped[obj], failures[obj], observedAt[obj], groundCache[obj] = nil, nil, nil, nil end
            elseif (skipped[obj] or 0) <= now and now - (observedAt[obj] or now) >= 0.18 then
                local part = getTouchPart(obj)
                if part and part.Parent and part.Transparency < 0.98 then
                    local directDistance = (part.Position - origin).Magnitude
                    if not maxDist or directDistance <= maxDist then
                        local ground = getCachedGround(obj, part)
                        -- Different map floors are valid targets; the raycast only rejects unsupported coins.
                        if ground then
                            table.insert(candidates, {
                                object = obj, part = part, ground = ground,
                                distance = directDistance, score = directDistance
                            })
                        else
                            skipped[obj] = now + 1.0
                        end
                    end
                else
                    table.remove(coins, i)
                    skipped[obj], observedAt[obj], groundCache[obj] = nil, nil, nil
                end
            end
        end
        if #candidates == 0 then return nil, nil end

        local best, bestScore
        for _, candidate in ipairs(candidates) do
            local position = candidate.part.Position
            local score = candidate.distance
            if method == "Under Map" then
                local hidden = Vector3.new(position.X, candidate.ground.Position.Y - 3.8, position.Z)
                score = (hidden - origin).Magnitude + math.max(0, position.Y - hidden.Y) * 1.2
            elseif method == "Teleport" then
                local nearest, second, third = math.huge, math.huge, math.huge
                for _, other in ipairs(candidates) do
                    if other ~= candidate then
                        local distance = (other.part.Position - position).Magnitude
                        if distance < nearest then
                            nearest, second, third = distance, nearest, second
                        elseif distance < second then
                            second, third = distance, second
                        elseif distance < third then
                            third = distance
                        end
                    end
                end
                if lastCollectedPosition and (position - lastCollectedPosition).Magnitude < 1000 then
                    -- Continue the previous collected cluster; fall back to the densest cluster on a cold start.
                    score = (position - lastCollectedPosition).Magnitude
                        + 0.25 * (math.min(nearest, 120) + math.min(second, 120))
                else
                    score = math.min(nearest, 300) + math.min(second, 300) + math.min(third, 300)
                end
            end

            if murdererPosition then
                local threatDistance = (position - murdererPosition).Magnitude
                if threatDistance < 24 then
                    score += CFG.FarmAvoidMurderer and 1000000 or 300
                elseif threatDistance < 48 then
                    score += (48 - threatDistance) * 2
                end
            end
            candidate.score = score
            if not bestScore or score < bestScore then best, bestScore = candidate, score end
        end

        if not best or (CFG.FarmAvoidMurderer and murdererPosition and bestScore >= 1000000) then
            return nil, nil
        end
        return best.object, best.distance
    end

    function API:GetStatus(origin, maxDist)
        if not running then return "OFF" end
        if not origin then return "WAITING" end
        if getActiveCoinContainer() ~= boundContainer then bindContainer() end
        if not boundContainer then return "WAIT MAP" end

        local eligible = 0
        local nearest = math.huge
        for _, obj in ipairs(coins) do
            if obj and obj.Parent and obj:IsDescendantOf(boundContainer) and isCoin(obj) then
                local part = getTouchPart(obj)
                if part and part.Parent and part.CanTouch and part.Transparency < 0.98 then
                    local ground = getCachedGround(obj, part)
                    if ground then
                        eligible += 1
                        nearest = math.min(nearest, (part.Position - origin).Magnitude)
                    end
                end
            end
        end

        if eligible == 0 then return "WAIT COINS" end
        if maxDist and nearest > maxDist then return "OUT OF RANGE" end
        return "READY"
    end

    function API:Skip(obj, seconds, confirmed)
        if not running or not obj then return end
        local duration = math.clamp(tonumber(seconds) or 1, 0.2, 30)
        if confirmed then
            failures[obj] = nil
        else
            local attempts = (failures[obj] or 0) + 1
            failures[obj] = attempts
            if attempts >= 4 then
                duration = 30
            else
                duration = math.max(duration, 0.6 * (2 ^ (attempts - 1)))
            end
        end
        skipped[obj] = os.clock() + duration
    end

    function API:Destroy()
        self:Stop()
    end
    return API
end)()

local farmBagCache = {checkedAt = 0, full = false, bag = nil}

-- PulseHub's collector listens for the game's local CoinCollected confirmation.
-- The event is used as a secondary success signal and to continue Teleport mode's coin cluster.
task.spawn(function()
    local remotes = ReplicatedStorage:WaitForChild("Remotes", 20)
    local gameplay = remotes and remotes:WaitForChild("Gameplay", 20)
    if not gameplay or not isRunning then return end
    local function listen(name, callback)
        local remote = gameplay:FindFirstChild(name)
        if remote and remote:IsA("RemoteEvent") then
            local connection = remote.OnClientEvent:Connect(callback)
            table.insert(farmRemoteConnections, connection)
            if name == "CoinCollected" then coinCollectedConnection = connection end
        end
    end
    local function resetRoundFarmState()
        farmLastCollectedPosition = nil
        farmCoinEventAt, farmCoinEventCount, farmTeleportLatency = 0, 0, 0.1
        farmTeleportHomeCFrame, farmTeleportPending = nil, nil
        table.clear(farmBagEventFull)
        farmBagCache.full, farmBagCache.checkedAt = false, 0
    end
    listen("RoundStart", resetRoundFarmState)
    listen("CoinsStarted", resetRoundFarmState)
    listen("RoundEndFade", resetRoundFarmState)
    listen("CoinCollected", function(kind, count, maximum)
        if not isRunning then return end
        farmCoinEventAt = os.clock()
        farmCoinEventCount += 1
        local character = LocalPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        farmLastCollectedPosition = root and root.Position or farmLastCollectedPosition
        local pending = farmTeleportPending
        if pending and root and pending.position
            and farmCoinEventAt >= pending.startedAt
            and farmCoinEventAt - pending.startedAt <= 1.5
            and (root.Position - pending.position).Magnitude <= 8 then
            local latencySample = farmCoinEventAt - pending.startedAt
            if latencySample > 0 and latencySample < 0.6 then
                farmTeleportLatency = math.clamp(farmTeleportLatency * 0.8 + latencySample * 0.2, 0.03, 0.3)
            end
            farmTeleportPending = nil
        end
        if type(kind) == "string" and type(count) == "number"
            and type(maximum) == "number" and maximum > 0 then
            farmBagEventFull[kind] = count >= maximum
            local anyType, allTypesFull = false, true
            for _, full in pairs(farmBagEventFull) do
                anyType = true
                if not full then allTypesFull = false end
            end
            farmBagCache.full = anyType and allTypesFull
            farmBagCache.checkedAt = farmCoinEventAt
        end
    end)
end)

local function isCoinBagFull()
    local now = os.clock()
    if now - farmBagCache.checkedAt < 0.45 then return farmBagCache.full end
    farmBagCache.checkedAt = now
    farmBagCache.full = false

    local anyType, allTypesFull = false, true
    for _, full in pairs(farmBagEventFull) do
        anyType = true
        if not full then allTypesFull = false end
    end
    if anyType and allTypesFull then
        farmBagCache.full = true
        return true
    end

    local main = PlayerGui and PlayerGui:FindFirstChild("MainGUI")
    local game = main and main:FindFirstChild("Game")
    local bag = game and game:FindFirstChild("CoinBags")
    farmBagCache.bag = bag
    if not bag or not bag.Visible then return false end

    for _, item in ipairs(bag:GetDescendants()) do
        if item.Name == "Full" and item:IsA("GuiObject")
            and not item:FindFirstAncestorWhichIsA("LuaSourceContainer") and item.Visible then
            farmBagCache.full = true
            break
        end
    end
    return farmBagCache.full
end

local glideInProgress = false
local function glideTo(target, speed, mode)
    local session = currentSession
    if glideInProgress or not session or session.destroyed or not session.root or not session.root.Parent
        or not session.humanoid or session.humanoid.Health <= 0 or not target or not target.Parent then
        return false
    end
    local character, root, humanoid = session.character, session.root, session.humanoid
    local startCFrame = root.CFrame
    local targetPart = getTouchPart(target)
    if not targetPart then return false end
    if mode == "farm" and (not coinTracker or not coinTracker:IsValid(target)
        or math.abs(targetPart.Position.Y - root.Position.Y) > 120) then return false end
    local farmMethod = mode == "farm" and CFG.FarmMethod or "Fast"
    if mode == "farm" then
        if farmMethod == "Under Map" then
            speed = math.clamp(tonumber(CFG.MaxSpeed) or 21, 8, 22)
        elseif farmMethod == "Teleport" then
            speed = math.clamp(tonumber(CFG.TeleportSpeed) or 25, 10, 26)
        else
            speed = math.clamp(tonumber(CFG.FastSpeed) or 24, 8, 24)
        end
    else
        speed = math.clamp(tonumber(speed) or 62, 25, 80)
    end

    local farmEventBaseline = mode == "farm" and farmCoinEventCount or 0
    local farmRouteStartedAt = os.clock()
    local oldAnchored = root.Anchored
    local oldVelocity = root.AssemblyLinearVelocity
    local oldAngularVelocity = root.AssemblyAngularVelocity
    local oldAutoRotate = humanoid.AutoRotate
    local collisionState, touchState = {}, {}
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            collisionState[part] = part.CanCollide
            touchState[part] = part.CanTouch
        end
    end
    local hipHeight = tonumber(humanoid.HipHeight) or 2
    -- Place the avatar's feet through the collectible's vertical band instead of stopping above it.
    local rootOffset = math.clamp(hipHeight + root.Size.Y * 0.5 - targetPart.Size.Y * 0.25, 2.2, 3.6)
    local farmGround = mode == "farm" and coinTracker and coinTracker:GetGround(target, targetPart) or nil
    local hidePosition = farmGround and Vector3.new(targetPart.Position.X, farmGround.Position.Y - 3.8, targetPart.Position.Z) or nil
    local routeStage = mode == "farm" and farmMethod == "Under Map" and hidePosition and "hide" or "contact"
    local routeDistance = (targetPart.Position + Vector3.new(0, rootOffset, 0) - root.Position).Magnitude
    if mode == "farm" and hidePosition then
        routeDistance = math.max(routeDistance, (hidePosition - root.Position).Magnitude + 4)
    end
    -- Pulse's movement budget scales with distance instead of cutting long cross-map routes at 22 seconds.
    local timeLimit = math.clamp(routeDistance / math.max(speed, 1) + 4.0, 4, 60)
    local arrived = false
    local collected = false
    local success, err
    local function restoreCollisionForPickup()
        for part, original in pairs(collisionState) do
            pcall(function() if part and part.Parent then part.CanCollide = original end end)
        end
        for part in pairs(touchState) do
            pcall(function() if part and part.Parent then part.CanTouch = true end end)
        end
    end
    local function targetWasCollected(part, wasTouchEnabled, wasTransparency)
        if not target or not target.Parent or not target:IsDescendantOf(Workspace) then return true end
        if target:GetAttribute("Collected") == true then return true end
        if mode == "farm" and farmCoinEventCount > farmEventBaseline
            and farmCoinEventAt >= farmRouteStartedAt
            and os.clock() - farmCoinEventAt <= 2
            and farmLastCollectedPosition and part and part.Parent
            and (farmLastCollectedPosition - part.Position).Magnitude <= 8 then
            return true
        end
        if part and part.Parent then
            if wasTouchEnabled and not part.CanTouch then return true end
            if wasTransparency < 0.98 and part.Transparency >= 0.98 then return true end
        end
        return false
    end
    local originalTargetCanTouch = targetPart.CanTouch
    local originalTargetTransparency = targetPart.Transparency
    glideInProgress = true
    success, err = pcall(function()
        -- Delta-safe smooth step movement: no anchoring; only temporary collision changes.
        root.Anchored = false
        humanoid.AutoRotate = false
        -- Clear inherited momentum once before stepping; clearing it every frame can fight replication.
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        for part in pairs(collisionState) do
            if part.Parent then
                part.CanCollide = false
                part.CanTouch = true
            end
        end
        -- Pulse-style Teleport waits against the last server-confirmed collection time,
        -- then snaps to the hidden point immediately before the next expected coin contact.
        if mode == "farm" and farmMethod == "Teleport" and hidePosition then
            local now = os.clock()
            if farmTeleportPending and now - farmTeleportPending.startedAt > 1.5 then
                farmTeleportPending = nil
            end
            local previous = farmLastCollectedPosition
            local eventAt = farmCoinEventAt
            if previous and eventAt > 0 and (targetPart.Position - previous).Magnitude < 1000 then
                local tpSpeed = math.clamp(tonumber(CFG.TeleportSpeed) or 25, 10, 26)
                local arrivalAt = eventAt + (targetPart.Position - previous).Magnitude / tpSpeed
                    - 0.4 * (farmTeleportLatency or 0.1)
                if now < arrivalAt - 0.35 then
                    if not farmTeleportHomeCFrame then
                        local homeCFrame
                        pcall(function()
                            local lobby = Workspace:FindFirstChild("Lobby")
                            local spawns = lobby and lobby:FindFirstChild("Spawns")
                            if not spawns then return end
                            local total, count = Vector3.zero, 0
                            for _, item in ipairs(spawns:GetDescendants()) do
                                if item:IsA("BasePart") then total += item.Position; count += 1 end
                            end
                            if count > 0 then
                                homeCFrame = CFrame.new(total / count + Vector3.new(0, 4, 0))
                            end
                        end)
                        farmTeleportHomeCFrame = homeCFrame or startCFrame
                    end
                    local home = farmTeleportHomeCFrame
                    if home and (root.Position - home.Position).Magnitude > 6 then
                        root.CFrame = home
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero
                    end
                    while isRunning and currentSession == session and not session.destroyed
                        and CFG.AutoFarm and os.clock() < arrivalAt - 0.35
                        and target.Parent and target:IsDescendantOf(Workspace)
                        and coinTracker and coinTracker:IsValid(target) do
                        RunService.Heartbeat:Wait()
                    end
                end
            end
            if isRunning and CFG.AutoFarm and target.Parent and target:IsDescendantOf(Workspace)
                and coinTracker and coinTracker:IsValid(target) then
                root.CFrame = CFrame.new(hidePosition, hidePosition + root.CFrame.LookVector)
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
            end
        end
        local startedAt = os.clock()
        local currentSpeed = 0
        local lastFarmTargetCheck = 0
        while isRunning and currentSession == session and not session.destroyed do
            if mode == "farm" and not CFG.AutoFarm then break end
            if mode == "pickup" and (CFG.AutoFarm or not CFG.AutoPickup) then break end
            if not root.Parent or humanoid.Health <= 0 or not target.Parent or not target:IsDescendantOf(Workspace) then break end
            targetPart = getTouchPart(target)
            if not targetPart or not targetPart.Parent then break end
            if mode == "farm" then
                if os.clock() - lastFarmTargetCheck >= 0.2 then
                    lastFarmTargetCheck = os.clock()
                    if not coinTracker or not coinTracker:IsValid(target) then break end
                end
                if targetPart.Transparency >= 0.98 then break end
            end

            local contactPosition = targetPart.Position + Vector3.new(0, rootOffset, 0)
            local destination = contactPosition
            if routeStage == "hide" then
                local ground = coinTracker and coinTracker:GetGround(target, targetPart) or farmGround
                if ground then
                    destination = Vector3.new(targetPart.Position.X, ground.Position.Y - 3.8, targetPart.Position.Z)
                else
                    routeStage = "contact"
                end
            end
            local delta = destination - root.Position
            local distance = delta.Magnitude
            if mode == "farm" and math.abs(delta.Y) > 120 then break end
            if distance <= 0.8 then
                if routeStage == "hide" then
                    routeStage = "contact"
                    currentSpeed = 0
                    continue
                end
                arrived = true
                -- Restore floor collisions before the collection attempt. The previous version
                -- kept the avatar non-collidable while trying to touch the coin, which could
                -- leave the feet below the hitbox and make the character hover beside it.
                restoreCollisionForPickup()
                local touchFn = GEN_ENV.LiberiumFireTouchInterest
                local touchParts = {}
                local function addTouchPart(part)
                    if part and part.Parent and part:IsA("BasePart") and not table.find(touchParts, part) then
                        pcall(function() part.CanTouch = true end)
                        table.insert(touchParts, part)
                    end
                end
                addTouchPart(root)
                if mode == "farm" then
                    -- Match PulseHub's broad character contact set: head, accessories, torso and feet.
                    addTouchPart(character:FindFirstChild("Head"))
                    for _, child in ipairs(character:GetChildren()) do
                        if child:IsA("Accessory") then addTouchPart(child:FindFirstChild("Handle")) end
                    end
                    addTouchPart(character:FindFirstChild("UpperTorso"))
                    addTouchPart(character:FindFirstChild("Torso"))
                    addTouchPart(character:FindFirstChild("LowerTorso"))
                    addTouchPart(character:FindFirstChild("LeftFoot") or character:FindFirstChild("Left Leg"))
                    addTouchPart(character:FindFirstChild("RightFoot") or character:FindFirstChild("Right Leg"))
                else
                    addTouchPart(character:FindFirstChild("Head"))
                    addTouchPart(character:FindFirstChild("UpperTorso"))
                    addTouchPart(character:FindFirstChild("Torso"))
                    addTouchPart(character:FindFirstChild("LowerTorso"))
                    addTouchPart(character:FindFirstChild("LeftFoot") or character:FindFirstChild("Left Leg"))
                    addTouchPart(character:FindFirstChild("RightFoot") or character:FindFirstChild("Right Leg"))
                end
                -- AutoFarm settles at one contact height instead of hopping through three heights.
                -- Keep AutoPickup's previous height fallback to avoid changing its behavior unnecessarily.
                local heightOffsets = mode == "farm"
                    -- Small contact-height retries cover R6/R15 rig and hitbox differences
                    -- without the large vertical hopping that made earlier farming look jerky.
                    and { rootOffset, math.clamp(rootOffset - 0.18, 2.0, 3.8), math.clamp(rootOffset + 0.18, 2.0, 3.8) }
                    or { rootOffset, math.max(2.0, rootOffset - 0.2), math.max(1.8, rootOffset - 0.45) }
                for _, heightOffset in ipairs(heightOffsets) do
                    if targetWasCollected(targetPart, originalTargetCanTouch, originalTargetTransparency) then collected = true break end
                    targetPart = getTouchPart(target)
                    if not targetPart or not targetPart.Parent then collected = true break end
                    local contact = targetPart.Position + Vector3.new(0, heightOffset, 0)
                    root.CFrame = CFrame.new(contact, contact + root.CFrame.LookVector)
                    if mode == "farm" and farmMethod == "Teleport" then
                        farmTeleportPending = {startedAt = os.clock(), position = targetPart.Position, target = target}
                    end
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                    -- Allow the avatar to settle at the contact point before checking pickup state.
                    RunService.Heartbeat:Wait()
                    if typeof(touchFn) == "function" then
                        -- A few heartbeat-separated probes let the server register different R6/R15 contact shapes.
                        local maxAttempts = 3
                        for attempt = 1, maxAttempts do
                            for _, bodyPart in ipairs(touchParts) do
                                pcall(function() touchFn(bodyPart, targetPart, 0) end)
                                pcall(function() touchFn(targetPart, bodyPart, 0) end)
                            end
                            RunService.Heartbeat:Wait()
                            for _, bodyPart in ipairs(touchParts) do
                                pcall(function() touchFn(bodyPart, targetPart, 1) end)
                                pcall(function() touchFn(targetPart, bodyPart, 1) end)
                            end
                            if targetWasCollected(targetPart, originalTargetCanTouch, originalTargetTransparency) then collected = true break end
                            if attempt < maxAttempts then task.wait(0.05) end
                        end
                    else
                        task.wait(mode == "farm" and 0.1 or 0.18)
                    end
                    if targetWasCollected(targetPart, originalTargetCanTouch, originalTargetTransparency) then collected = true break end
                    if mode ~= "farm" then task.wait(0.08) end
                end
                -- Never leave the player parked inside the same uncollectable hitbox.
                if not collected and root.Parent and targetPart and targetPart.Parent then
                    local away = root.Position + Vector3.new(0, 0.8, 2.5)
                    root.CFrame = CFrame.new(away, away + root.CFrame.LookVector)
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                end
                break
            end
            if os.clock() - startedAt > timeLimit then break end
            local frameDt = RunService.Heartbeat:Wait()
            local dt = math.clamp(frameDt, 1 / 240, 0.06)
            -- Accelerate into the route, then brake gently in the last few studs.
            currentSpeed = math.min(speed, currentSpeed + 155 * dt)
            local brakingSpeed = math.max(14, distance * 7)
            local step = math.min(math.min(currentSpeed, brakingSpeed) * dt, distance)
            if distance > 0 then
                local newPosition = root.Position + delta.Unit * step
                root.CFrame = CFrame.new(newPosition, newPosition + root.CFrame.LookVector)
            end
        end
    end)

    if mode == "farm" and targetWasCollected(targetPart, originalTargetCanTouch, originalTargetTransparency) then
        collected = true
        arrived = true
    end

    -- On an interrupted farm route, return smoothly to the exact starting position
    -- before restoring collisions; never leave the avatar suspended over an unsupported gap.
    if mode == "farm" and not arrived and root and root.Parent and humanoid
        and humanoid.Parent and humanoid.Health > 0 and LocalPlayer.Character == character then
        pcall(function()
            local from = root.Position
            local destination = startCFrame.Position
            for stepIndex = 1, 7 do
                if not root.Parent or humanoid.Health <= 0 then break end
                local alpha = stepIndex / 7
                local position = from:Lerp(destination, alpha)
                root.CFrame = CFrame.new(position, position + startCFrame.LookVector)
                RunService.Heartbeat:Wait()
            end
        end)
    end

    -- Always restore physics flags, even when the target disappears or the executor errors.
    pcall(function()
        if root and root.Parent then
            root.Anchored = oldAnchored
            root.AssemblyLinearVelocity = arrived and Vector3.zero or oldVelocity
            root.AssemblyAngularVelocity = arrived and Vector3.zero or oldAngularVelocity
        end
    end)
    pcall(function() if humanoid and humanoid.Parent then humanoid.AutoRotate = oldAutoRotate end end)
    for part, original in pairs(collisionState) do
        pcall(function() if part and part.Parent then part.CanCollide = original end end)
    end
    for part, original in pairs(touchState) do
        pcall(function() if part and part.Parent then part.CanTouch = original end end)
    end
    glideInProgress = false
    if not success then warn("[Liberium AutoFarm] movement recovered from error: " .. tostring(err)) end
    return success == true and collected
end


local function tryDistanceGunPickup(target, session)
    if not session or session.destroyed or not session.root or not session.root.Parent
        or not target or not target.Parent or not weaponTracker or not weaponTracker:IsValid(target) then
        return false, "invalid"
    end

    local touchFn = GEN_ENV.LiberiumFireTouchInterest
    local root = session.root
    local character = session.character
    local targetPart = weaponTracker:GetTouchPart(target)
    if not targetPart or not targetPart.Parent then return false, "invalid" end
    if (targetPart.Position - root.Position).Magnitude > CFG.PickupRange then return false, "out-of-range" end

    local function hasGun()
        local currentCharacter = LocalPlayer.Character
        local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
        return (currentCharacter and currentCharacter:FindFirstChild("Gun") ~= nil)
            or (backpack and backpack:FindFirstChild("Gun") ~= nil)
    end

    -- Arceus X builds may not expose firetouchinterest. In that case use the
    -- existing smooth contact route so native character/part touch can collect the drop.
    if typeof(touchFn) ~= "function" then
        local routed = glideTo(target, math.max(25, tonumber(CFG.FastSpeed) or 24), "pickup")
        task.wait(0.12)
        if hasGun() or not weaponTracker:IsValid(target) then return true, "collected" end
        return routed == true, "fallback-unconfirmed"
    end

    local touchParts = {}
    local function addTouchPart(part)
        if part and part.Parent and part:IsA("BasePart") and not table.find(touchParts, part) then
            table.insert(touchParts, part)
        end
    end

    addTouchPart(root)
    addTouchPart(character:FindFirstChild("Head"))
    addTouchPart(character:FindFirstChild("RightHand") or character:FindFirstChild("Right Arm"))
    addTouchPart(character:FindFirstChild("LeftHand") or character:FindFirstChild("Left Arm"))
    addTouchPart(character:FindFirstChild("UpperTorso"))
    addTouchPart(character:FindFirstChild("Torso"))
    addTouchPart(character:FindFirstChild("LowerTorso"))
    addTouchPart(character:FindFirstChild("RightFoot") or character:FindFirstChild("Right Leg"))
    addTouchPart(character:FindFirstChild("LeftFoot") or character:FindFirstChild("Left Leg"))

    local function fire(first, second, state)
        local ok = pcall(touchFn, first, second, state)
        return ok
    end

    for attempt = 1, 5 do
        if not isRunning or currentSession ~= session or session.destroyed
            or not root.Parent or not session.humanoid or session.humanoid.Health <= 0 then
            return false, "interrupted"
        end
        if hasGun() or not weaponTracker:IsValid(target) then return true, "collected" end

        targetPart = weaponTracker:GetTouchPart(target)
        if not targetPart or not targetPart.Parent then return true, "collected" end
        if (targetPart.Position - root.Position).Magnitude > CFG.PickupRange then return false, "out-of-range" end

        for _, bodyPart in ipairs(touchParts) do
            if bodyPart.Parent then
                fire(bodyPart, targetPart, 0)
                fire(targetPart, bodyPart, 0)
            end
        end
        RunService.Heartbeat:Wait()
        for _, bodyPart in ipairs(touchParts) do
            if bodyPart.Parent then
                fire(bodyPart, targetPart, 1)
                fire(targetPart, bodyPart, 1)
            end
        end

        if hasGun() or not weaponTracker:IsValid(target) then return true, "collected" end
        if attempt < 5 then task.wait(0.1) end
    end

    return false, "unconfirmed"
end

-- LIBERIUM API v92: additive developer facade.
-- Keeps the v91.18 UI and gameplay systems intact while providing a consistent
-- registry, event bus, middleware hooks, pre-checks, and cleanup lifecycle.
local LiberiumAPI = {}
do
    local api = { _win = win, _elements = {}, _events = {}, _hooks = {}, _unload = {}, _destroyed = false }
    local function normalizeId(value)
        return string.lower(tostring(value or "")):gsub("[^%w_%-]", "_")
    end
    local function track(obj)
        api._elements[normalizeId(obj.id)] = obj
        return obj
    end
    local function makeElement(id, control, kind)
        local obj = { id = id, kind = kind, control = control, _pre = {}, _opts = nil }
        function obj:get()
            if self.kind == "toggle" then return self._value end
            if self.kind == "slider" or self.kind == "dropdown" then return self._value end
            return self._value
        end
        function obj:Set(value, silent)
            if self.control and self.control.Set then self.control:Set(value, silent) end
            self._value = value
            return self
        end
        function obj:set(value, silent) return self:Set(value, silent) end
        function obj:Get() return self:get() end
        function obj:on(callback) self._callback = callback return self end
        function obj:pre(callback)
            if type(callback) == "function" then table.insert(self._pre, callback) end
            return self
        end
        function obj:opts()
            if self._opts then return self._opts end
            local sub = api._win.Tabs[#api._win.Tabs]
            local activeSub = sub and sub.ActiveSub
            if not activeSub then return nil end
            local sec = activeSub:AddSection(tostring(self.id) .. " OPTIONS")
            self._opts = api._makeSection(sec, tostring(self.id) .. "/options")
            return self._opts
        end
        return track(obj)
    end
    local function bindCallback(obj, callback)
        return function(value)
            for _, check in ipairs(obj._pre) do
                local ok, result = pcall(check, value)
                if not ok or result == false then return end
            end
            obj._value = value
            if obj._callback then task.spawn(obj._callback, value) end
            if callback then callback(value) end
            api:emit("change", obj.id, value)
        end
    end
    function api._makeSection(section, prefix)
        local wrapper = { _section = section, _prefix = prefix }
        local function idFor(config) return normalizeId(prefix .. "/" .. tostring(config.name or "element")) end
        function wrapper:toggle(config)
            config = config or {}
            local id = idFor(config)
            local obj
            local cb = config.callback
            local control = self._section:AddToggle(config.name or id, config.default == true or config.opts == true, function(v)
                if obj then bindCallback(obj, cb)(v) end
            end)
            obj = makeElement(id, control, "toggle")
            obj._value = config.default == true or config.opts == true
            return obj
        end
        function wrapper:slider(config)
            config = config or {}
            local id = idFor(config)
            local obj
            local control = self._section:AddSlider(config.name or id, tonumber(config.min) or 0, tonumber(config.max) or 100, tonumber(config.default) or 0, function(v)
                if obj then bindCallback(obj, config.callback)(v) end
            end, tonumber(config.step) or 1)
            obj = makeElement(id, control, "slider")
            obj._value = tonumber(config.default) or 0
            return obj
        end
        function wrapper:dropdown(config)
            config = config or {}
            local id = idFor(config)
            local obj
            local control = self._section:AddDropdown(config.name or id, config.list or {}, config.default, function(v)
                if obj then bindCallback(obj, config.callback)(v) end
            end)
            obj = makeElement(id, control, "dropdown")
            obj._value = config.default
            return obj
        end
        function wrapper:button(config)
            config = config or {}
            local id = idFor(config)
            local obj
            local control = self._section:AddButton(config.name or id, function()
                if obj then
                    for _, check in ipairs(obj._pre) do
                        local ok, result = pcall(check, true)
                        if not ok or result == false then return end
                    end
                end
                if config.callback then task.spawn(config.callback) end
                api:emit("action", id)
            end)
            obj = makeElement(id, control, "button")
            return obj
        end
        function wrapper:label(config)
            config = config or {}
            local id = idFor(config)
            local obj = makeElement(id, nil, "label")
            obj._value = config.text or config.name or ""
            return obj
        end
        return wrapper
    end
    function api:tab(config, icon)
        local name = type(config) == "table" and config.name or config
        local glyph = type(config) == "table" and config.icon or icon
        local tab = self._win:AddTab(tostring(name or "Tab"), glyph)
        local wrapper = { _tab = tab }
        function wrapper:sec(sectionConfig)
            local sectionName = type(sectionConfig) == "table" and sectionConfig.name or sectionConfig
            local sub = self._tab:AddSubTab(tostring(sectionName or "Section"))
            local sec = sub:AddSection(tostring(sectionName or "Section"))
            return api._makeSection(sec, normalizeId(name) .. "/" .. normalizeId(sectionName))
        end
        function wrapper:sub(subConfig)
            local subName = type(subConfig) == "table" and subConfig.name or subConfig
            local sub = self._tab:AddSubTab(tostring(subName or "Sub"))
            local subWrapper = { _sub = sub }
            function subWrapper:sec(secConfig)
                local secName = type(secConfig) == "table" and secConfig.name or secConfig
                return api._makeSection(self._sub:AddSection(tostring(secName or "Section")), normalizeId(name) .. "/" .. normalizeId(subName) .. "/" .. normalizeId(secName))
            end
            return subWrapper
        end
        return wrapper
    end
    function api:el(id) return self._elements[normalizeId(id)] end
    function api:on(eventName, callback)
        if type(callback) ~= "function" then return nil end
        local bucket = self._events[eventName] or {}
        self._events[eventName] = bucket
        local entry = { callback = callback, connected = true }
        table.insert(bucket, entry)
        local connection = {}
        function connection:Disconnect() entry.connected = false end
        return connection
    end
    function api:emit(eventName, ...)
        local bucket = self._events[eventName]
        if not bucket then return end
        for _, entry in ipairs(bucket) do
            if entry.connected then task.spawn(entry.callback, ...) end
        end
    end
    function api:hook(name, middleware)
        if type(middleware) ~= "function" then return false end
        local bucket = self._hooks[name] or {}
        self._hooks[name] = bucket
        table.insert(bucket, middleware)
        return true
    end
    function api:dispatchHook(name, original, ...)
        local args = table.pack(...)
        local function invoke(index, ...)
            local middleware = self._hooks[name] and self._hooks[name][index]
            if not middleware then return original(...) end
            return middleware(function(...) return invoke(index + 1, ...) end, ...)
        end
        return invoke(1, table.unpack(args, 1, args.n))
    end
    function api:unload(callback)
        if type(callback) == "function" then table.insert(self._unload, callback) end
    end
    function api:destroy()
        if self._destroyed then return end
        self._destroyed = true
        for _, callback in ipairs(self._unload) do pcall(callback) end
        for _, bucket in pairs(self._events) do
            for _, entry in ipairs(bucket) do entry.connected = false end
        end
        table.clear(self._elements)
        table.clear(self._hooks)
    end
    -- Public API accepts both Shitaro-style dot calls (sh.tab(...)) and colon calls.
    local function dualCall(method)
        return function(first, ...)
            if first == api then return method(api, ...) end
            return method(api, first, ...)
        end
    end
    api.tab = dualCall(api.tab)
    api.el = dualCall(api.el)
    api.on = dualCall(api.on)
    api.emit = dualCall(api.emit)
    api.hook = dualCall(api.hook)
    api.dispatchHook = dualCall(api.dispatchHook)
    api.unload = dualCall(api.unload)
    LiberiumAPI = api
end

    while isRunning and homeFarmValue and homeFarmValue.Parent do
        local good = THEME.Good
        local muted = THEME.SubText
        local waitColor = Color3.fromRGB(255,195,90)

        if CFG.AutoFarm then
            if not canFarmNow then
                setHomeValue(homeFarmValue,"WAITING",waitColor)
            else
                local session = currentSession
                if not session or session.destroyed or not session.root or not session.root.Parent then
                    setHomeValue(homeFarmValue,"WAITING",waitColor)
                else
                    local farmStatus = coinTracker:GetStatus(session.root.Position,nil)
                    local farmColor = farmStatus == "READY" and good
                        or farmStatus == "OUT OF RANGE" and waitColor
                        or farmStatus == "WAIT MAP" and waitColor
                        or farmStatus == "WAIT COINS" and waitColor
                        or muted
                    setHomeValue(homeFarmValue,farmStatus,farmColor)
                end
            end
        else
            setHomeValue(homeFarmValue,"OFF",muted)
        end

        if CFG.AutoPickup then
            if CFG.AutoFarm then
                setHomeValue(homePickupValue,"PAUSED",muted)
            elseif not canFarmNow then
                setHomeValue(homePickupValue,"WAITING",waitColor)
            else
                local session = currentSession
                if not session or session.destroyed or not session.root or not session.root.Parent then
                    setHomeValue(homePickupValue,"WAITING",waitColor)
                else
                    local pickupStatus = weaponTracker:GetStatus(session.root.Position,CFG.PickupRange)
                    local pickupColor = pickupStatus == "READY" and good
                        or pickupStatus == "OUT OF RANGE" and waitColor
                        or pickupStatus == "NO DROP" and waitColor
                        or muted
                    setHomeValue(homePickupValue,pickupStatus,pickupColor)
                end
            end
        else
            setHomeValue(homePickupValue,"OFF",muted)
        end

        local espActive = CFG.ESP_Outline or CFG.ESP_Fill or CFG.ESP_Name or CFG.ESP_Distance or CFG.SkeletonESP or CFG.LootESP
        setHomeValue(homeEspValue,espActive and "ON" or "OFF",espActive and THEME.Accent2 or muted)

        local map = getActiveMap()
        if map and map:FindFirstChild("CoinContainer") then
            setHomeValue(homeMapValue,"ROUND",good)
        elseif map and map:FindFirstChild("Spawns") then
            setHomeValue(homeMapValue,"LOBBY / WAIT",waitColor)
        else
            setHomeValue(homeMapValue,"LOADING",muted)
        end

        task.wait(0.75)
    end
end)
