-- Run this file with the executor's loadstring entry:
-- loadstring(game:HttpGet("https://raw.githubusercontent.com/Liberator481993/LiberiumHub/liberium-v91-80-arceus-diagnostics/Liberium_Loader_v91.80.txt"))()

local Players = game:GetService("Players")
local player = Players.LocalPlayer
if not player then
    warn("[Liberium Loader] LocalPlayer unavailable.")
    return
end
local playerGui = player:FindFirstChildOfClass("PlayerGui") or player:WaitForChild("PlayerGui", 12)
if not playerGui then
    warn("[Liberium Loader] PlayerGui unavailable.")
    return
end

local old = playerGui:FindFirstChild("LiberiumLoaderDiagnostic")
if old then pcall(function() old:Destroy() end) end

local gui = Instance.new("ScreenGui")
gui.Name = "LiberiumLoaderDiagnostic"
gui.ResetOnSpawn = false
gui.IgnoreGuiInset = true
gui.DisplayOrder = 32760
gui.Parent = playerGui

local panel = Instance.new("Frame")
panel.AnchorPoint = Vector2.new(0.5, 0)
panel.Position = UDim2.new(0.5, 0, 0, 42)
panel.Size = UDim2.new(0.9, 0, 0, 56)
panel.BackgroundColor3 = Color3.fromRGB(13, 17, 29)
panel.BackgroundTransparency = 0.04
panel.BorderSizePixel = 0
panel.Parent = gui
Instance.new("UICorner", panel).CornerRadius = UDim.new(0, 10)
local outline = Instance.new("UIStroke", panel)
outline.Color = Color3.fromRGB(151, 111, 255)
outline.Transparency = 0.2

local label = Instance.new("TextLabel")
label.BackgroundTransparency = 1
label.Position = UDim2.fromOffset(10, 5)
label.Size = UDim2.new(1, -20, 1, -10)
label.Font = Enum.Font.GothamSemibold
label.TextSize = 12
label.TextColor3 = Color3.fromRGB(232, 238, 255)
label.TextWrapped = true
label.TextXAlignment = Enum.TextXAlignment.Center
label.TextYAlignment = Enum.TextYAlignment.Center
label.Text = "LIBERIUM v91.80 • LOADER STARTED"
label.Parent = panel

local function status(message, color)
    local text = "[Liberium Loader] " .. tostring(message)
    warn(text)
    pcall(function()
        label.Text = "LIBERIUM v91.80 • " .. tostring(message)
        if color then outline.Color = color end
    end)
end

local sourceUrl = "https://raw.githubusercontent.com/Liberator481993/LiberiumHub/liberium-v91-80-arceus-diagnostics/Liberium_Hub_v91.80_Arceus.txt"
status("DOWNLOADING SCRIPT", Color3.fromRGB(77, 220, 246))

local downloadOk, source = pcall(function()
    return game:HttpGet(sourceUrl)
end)
if not downloadOk then
    status("DOWNLOAD ERROR: " .. tostring(source), Color3.fromRGB(255, 85, 100))
    return
end
if type(source) ~= "string" or #source < 50000 then
    status("BAD RESPONSE / FILE TOO SHORT (" .. tostring(type(source) == "string" and #source or 0) .. " BYTES)", Color3.fromRGB(255, 85, 100))
    return
end
status("DOWNLOADED " .. tostring(#source) .. " BYTES • COMPILING", Color3.fromRGB(77, 220, 246))

if type(loadstring) ~= "function" then
    status("LOADSTRING NOT AVAILABLE IN THIS EXECUTOR", Color3.fromRGB(255, 85, 100))
    return
end
local compiled, compileError = loadstring(source)
if type(compiled) ~= "function" then
    status("COMPILE ERROR: " .. tostring(compileError), Color3.fromRGB(255, 85, 100))
    return
end

status("COMPILED • STARTING HUB", Color3.fromRGB(93, 232, 177))
local ran, runtimeError = pcall(compiled)
if not ran then
    status("RUNTIME ERROR: " .. tostring(runtimeError), Color3.fromRGB(255, 85, 100))
    return
end

status("HUB CHUNK RETURNED • CHECK MENU / SEND THIS STATUS", Color3.fromRGB(93, 232, 177))
task.delay(10, function()
    if gui and gui.Parent then pcall(function() gui:Destroy() end) end
end)
